10public class LockFreeQueue<T> {
11    private static class Node<T> {
12        T value;
13 AtomicReference<Node<T>> next;
14        Node(T value, Node<T> next) {
15            this.value = value;
16            this.next = new AtomicReference<>(next);
17        }
18    }
19
20    private final AtomicReference<Node<T>> head, tail;
21    private final AtomicInteger size = new AtomicInteger(0);
22
23    public LockFreeQueue() {
24        Node<T> dummy = new Node<>(null, null);
25        head = new AtomicReference<>(dummy);
26        tail = new AtomicReference<>(dummy);
27    }
28
29    // TODO: 实现enqueue
30    public void enqueue(T value) {
31        Node<T> newNode = new Node<>(value, null);
32        while (true) {
33            Node<T> curTail = tail.get();
34            Node<T> next = curTail.next.get();
35            if (curTail == tail.get()) {
36                if (next == null) {
37                    if (curTail.next.compareAndSet(null, newNode)) {
38                        tail.compareAndSet(curTail, newNode);
39                        size.incrementAndGet();
40                        return;
41                    }
42                } else {
43                    tail.compareAndSet(curTail, next);
44                }
45            }
46        }
47    }
